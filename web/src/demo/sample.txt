import { verifySignature } from '../../delivery/verify-signature';
import { registerEndpoint } from '../../delivery/endpoint-registry';

export async function subscribe(req: Request) {
  const body = await req.text();
  const signature = req.headers.get('x-relay-signature') ?? '';
  if (!verifySignature(body, signature)) {
    return new Response('Invalid signature', { status: 401 });
  }
  const endpoint = await registerEndpoint(JSON.parse(body));
  return Response.json(endpoint, { status: 201 });
}

@@ server/delivery/retry/backoff.ts
import type { RetryPolicy } from '../model/types';
import { tenantPolicy } from '../tenant-config';

const BASE_MS = 1_000;
const CAP_MS = 15 * 60_000;

export function backoffDelay(attempt: number, policy: RetryPolicy = tenantPolicy()) {
  const exp = Math.min(CAP_MS, BASE_MS * 2 ** attempt * policy.multiplier);
  return Math.round(exp / 2 + Math.random() * (exp / 2));
}

@@ server/delivery/pipeline/sign-request.ts
import { createHmac } from 'node:crypto';
import { currentSecret } from '../secrets';

export function signRequest(body: string) {
  const ts = Math.floor(Date.now() / 1000);
  const mac = createHmac('sha256', currentSecret()).update(`${ts}.${body}`).digest('hex');
  return { body, headers: { 'x-relay-signature': `t=${ts},v1=${mac}` } };
}

@@ server/delivery/http-client.ts
export const httpClient = {
  async post(url: string, body: string, opts: { headers: Record<string, string>; timeoutMs: number }) {
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(), opts.timeoutMs);
    try {
      const res = await fetch(url, { method: 'POST', body, headers: opts.headers, signal: ctrl.signal });
      return { status: res.status };
    } catch (e) {
      return { status: 0, error: String(e) };
    } finally {
      clearTimeout(timer);
    }
  },
};
@@ server/delivery/rate-limit.ts
import { tenantById } from './tenant-config';
import type { RateLimit } from './model/types';

const buckets = new Map<string, { tokens: number; at: number }>();
const DEFAULT: RateLimit = { perSecond: 10, burst: 20 };

export async function allow(tenant: string) {
  const b = buckets.get(tenant) ?? { tokens: DEFAULT.burst, at: Date.now() };
  const limit = { ...DEFAULT, ...tenantById(tenant)?.limits };
  b.tokens = Math.min(limit.burst, b.tokens + ((Date.now() - b.at) / 1000) * limit.perSecond);
  b.at = Date.now();
  if (b.tokens < 1) await new Promise((r) => setTimeout(r, 1000 / limit.perSecond));
  buckets.set(tenant, { ...b, tokens: b.tokens - 1 });
}
@@ server/delivery/verify-signature.ts
import { createHmac, timingSafeEqual } from 'node:crypto';
import { secretsFor } from './secrets';

export function verifySignature(body: string, header: string) {
  const { t, v1 } = Object.fromEntries(header.split(',').map((p) => p.split('=')));
  const expected = secretsFor().map((s) => createHmac('sha256', s).update(`${t}.${body}`).digest('hex'));
  return expected.some((mac) => mac === v1);
}

@@ server/delivery/payload-schema.ts
const MAX_BYTES = 256 * 1024;

export type Payload = { id: string; type: string; data: unknown; sentAt: string };

export function validatePayload(p: Payload) {
  if (!p.id || !p.type) throw new Error('Payload needs an id and a type');
  if (!/^[a-z]+(\.[a-z_]+)+$/.test(p.type)) throw new Error(`Bad event type: ${p.type}`);
  const size = new TextEncoder().encode(JSON.stringify(p)).length;
  if (size > MAX_BYTES) throw new Error(`Payload is ${size} bytes, limit is ${MAX_BYTES}`);
  return p;
}

@@ server/delivery/circuit-breaker.ts
import { healthOf } from './endpoint-health';
import { tenantById } from './tenant-config';

const openUntil = new Map<string, number>();

export function isOpen(endpointId: string) {
  const until = openUntil.get(endpointId) ?? 0;
  if (until > Date.now()) return true;
  if (healthOf(endpointId).failures < 5) return false;
  openUntil.set(endpointId, Date.now() + 60_000);
  return true;
}
